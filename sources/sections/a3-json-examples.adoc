== Examples in JSON

[source,json]
----
{
  "givenName": [
    {
    }
  ],
  "generationalName": [
    {
      "abbreviated": "III"
      // 三世
      // The Third
    }
  ],
  "honorific": [
    {
      "type": "academic",
      "content": "Dr.",
    }
  ],
  "template": {
    "lang": "en",
    "body": [
      {
        "key": "surname"
      },
      {
        "key": "givenName",
        "options": [
          "abbreviationPriority", 3
        ]
      },
      {
        "key": "surname"
      },
      "generational"
    ]
  }
}
----

  "affiliation": "https://nist.gov/directory/harold.booth/affiliation"

  "surname": "Booth",
